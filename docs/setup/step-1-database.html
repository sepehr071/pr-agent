Database Setup

Create a Supabase project (or use your own PostgreSQL) and run the schema migrations.

1.1 Create Supabase Project

  1. Go to https://supabase.com and sign up (or use your own PostgreSQL instance)
  2. Click "New Project" — name it gitlab-reviews, set a strong database password, pick the closest region
  3. Wait 2–3 minutes for provisioning to complete

1.2 Create Base Table

Open SQL Editor in Supabase and run this query:

SQL — Base Table
-- Create table for storing pending reviews CREATE TABLE pending_reviews ( id SERIAL PRIMARY KEY, review_id VARCHAR(255) UNIQUE NOT NULL, mr_iid INTEGER NOT NULL, project_id INTEGER NOT NULL, findings JSONB NOT NULL, stats JSONB, draft_note_id INTEGER, base_sha VARCHAR(64), head_sha VARCHAR(64), start_sha VARCHAR(64), status VARCHAR(50) DEFAULT 'pending_approval', approved_by VARCHAR(255), approved_at TIMESTAMP, rejected_by VARCHAR(255), rejected_at TIMESTAMP, created_at TIMESTAMP DEFAULT NOW() ); -- Create indexes for fast lookups CREATE INDEX idx_pending_reviews_review_id ON pending_reviews(review_id); CREATE INDEX idx_pending_reviews_status ON pending_reviews(status); CREATE INDEX idx_pending_reviews_mr ON pending_reviews(project_id, mr_iid);

You should see "Success. No rows returned".

1.3 Run v3.0 Migration

This adds the learning system — 5 new tables, helper functions, and a dashboard view. Run this in the same SQL Editor:

SQL — v3.0 Migration (6 tables, 2 functions, 1 view)
-- ===================================================== -- GitLab MR Review System - Database Migration v3.0 -- ===================================================== CREATE EXTENSION IF NOT EXISTS pgcrypto; -- Add new columns to pending_reviews ALTER TABLE pending_reviews ADD COLUMN IF NOT EXISTS context_data JSONB DEFAULT '{}', ADD COLUMN IF NOT EXISTS file_analysis JSONB DEFAULT '{}', ADD COLUMN IF NOT EXISTS confidence_score DECIMAL(5,2), ADD COLUMN IF NOT EXISTS review_metadata JSONB DEFAULT '{}', ADD COLUMN IF NOT EXISTS custom_comments JSONB DEFAULT '[]'; CREATE INDEX IF NOT EXISTS idx_pending_reviews_confidence ON pending_reviews(confidence_score) WHERE confidence_score IS NOT NULL; CREATE INDEX IF NOT EXISTS idx_pending_reviews_metadata ON pending_reviews USING GIN(review_metadata); -- Review History (Learning System) CREATE TABLE IF NOT EXISTS review_history ( id SERIAL PRIMARY KEY, project_id INTEGER NOT NULL, mr_iid INTEGER NOT NULL, file_path VARCHAR(512) NOT NULL, issue_type VARCHAR(100) NOT NULL, issue_subcategory VARCHAR(100), pattern_hash VARCHAR(64), severity VARCHAR(20), finding_data JSONB, was_accepted BOOLEAN, was_fixed BOOLEAN DEFAULT NULL, created_at TIMESTAMP DEFAULT NOW(), reviewed_at TIMESTAMP, review_id VARCHAR(255) ); CREATE INDEX IF NOT EXISTS idx_review_history_project ON review_history(project_id, file_path); CREATE INDEX IF NOT EXISTS idx_review_history_pattern ON review_history(pattern_hash) WHERE pattern_hash IS NOT NULL; CREATE INDEX IF NOT EXISTS idx_review_history_type ON review_history(issue_type, issue_subcategory); CREATE INDEX IF NOT EXISTS idx_review_history_accepted ON review_history(was_accepted) WHERE was_accepted IS NOT NULL; -- Project Patterns (Context Awareness) CREATE TABLE IF NOT EXISTS project_patterns ( id SERIAL PRIMARY KEY, project_id INTEGER NOT NULL UNIQUE, patterns JSONB NOT NULL, last_updated TIMESTAMP DEFAULT NOW(), total_reviews INTEGER DEFAULT 0, accepted_findings INTEGER DEFAULT 0, rejected_findings INTEGER DEFAULT 0 ); -- Ignored Patterns (Auto-Ignore) CREATE TABLE IF NOT EXISTS ignored_patterns ( id SERIAL PRIMARY KEY, project_id INTEGER NOT NULL, pattern_type VARCHAR(100) NOT NULL, pattern_regex VARCHAR(512), added_by VARCHAR(255), reason TEXT, is_active BOOLEAN DEFAULT TRUE, created_at TIMESTAMP DEFAULT NOW(), UNIQUE(project_id, pattern_type) ); -- Agent Runs (Performance Tracking) CREATE TABLE IF NOT EXISTS agent_runs ( id SERIAL PRIMARY KEY, review_id VARCHAR(255) NOT NULL, agent_name VARCHAR(100) NOT NULL, started_at TIMESTAMP DEFAULT NOW(), completed_at TIMESTAMP, duration_ms INTEGER, tokens_used INTEGER, findings_count INTEGER, error_message TEXT, status VARCHAR(50) DEFAULT 'running' ); -- CTO Feedback (Continuous Improvement) CREATE TABLE IF NOT EXISTS cto_feedback ( id SERIAL PRIMARY KEY, review_id VARCHAR(255) NOT NULL, project_id INTEGER NOT NULL, mr_iid INTEGER NOT NULL, feedback_type VARCHAR(100) NOT NULL, feedback TEXT NOT NULL, severity VARCHAR(20), created_at TIMESTAMP DEFAULT NOW() ); -- Helper: Pattern Hash CREATE OR REPLACE FUNCTION calculate_pattern_hash(code_snippet TEXT, issue_type VARCHAR) RETURNS VARCHAR(64) AS $$ BEGIN RETURN encode(digest(issue_type || ':' || substring(code_snippet, 1, 1000), 'sha256'), 'hex'); END; $$ LANGUAGE plpgsql IMMUTABLE; -- Helper: Project Stats CREATE OR REPLACE FUNCTION get_project_stats(p_project_id INTEGER) RETURNS JSONB AS $$ DECLARE v_stats JSONB; BEGIN SELECT jsonb_build_object( 'total_reviews', COUNT(DISTINCT rh.review_id), 'accepted_findings', COUNT(*) FILTER (WHERE rh.was_accepted = TRUE), 'rejected_findings', COUNT(*) FILTER (WHERE rh.was_accepted = FALSE), 'most_common_issues', ( SELECT jsonb_agg(jsonb_build_object('type', issue_type, 'count', cnt)) FROM ( SELECT issue_type, COUNT(*) as cnt FROM review_history WHERE project_id = p_project_id GROUP BY issue_type ORDER BY cnt DESC LIMIT 5 ) sub ) ) INTO v_stats FROM review_history rh WHERE rh.project_id = p_project_id; RETURN COALESCE(v_stats, '{}'::jsonb); END; $$ LANGUAGE plpgsql; -- Dashboard View CREATE OR REPLACE VIEW active_reviews_dashboard AS SELECT pr.review_id, pr.mr_iid, pr.project_id, pr.status, pr.created_at, pr.confidence_score, (pr.review_metadata->>'risk_score')::INTEGER AS risk_score, (pr.review_metadata->>'executive_summary')::TEXT AS executive_summary, (pr.stats->>'total')::INTEGER AS total_findings, (pr.stats->>'critical')::INTEGER AS critical_findings, (pr.stats->>'high')::INTEGER AS high_findings, pr.approved_by FROM pending_reviews pr WHERE pr.status IN ('pending_approval', 'changes_requested') ORDER BY (pr.stats->>'critical')::INTEGER DESC, pr.created_at ASC;
